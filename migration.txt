-- ============================================================================
-- MIGRATION : operationorder → workshop
-- ============================================================================
-- Ce fichier contient les requetes SQL de migration des donnees
-- de l'ancien module operationorder vers le module workshop.
--
-- A executer dans l'ordre, requete par requete.
-- Prerequis : les tables workshop (llx_workshop_*) doivent deja exister.
-- ============================================================================

-- ============================================================================
-- ETAPE 1 : Migration des statuts OR
-- ============================================================================
-- Strategie :
--   - Les anciens statuts (llx_operationorder_status) sont par entite (entity variable)
--   - Les nouveaux statuts (llx_workshop_operationorder_status) sont transverses (entity = 0)
--   - On insere un statut par code unique (GROUP BY code)
--   - On cree ensuite une table de correspondance ancien rowid -> nouveau rowid via le code
-- ============================================================================

-- 1.1 : Vidage de la table de destination avant insertion
TRUNCATE TABLE llx_workshop_operationorder_status;

-- 1.2 : Insertion des statuts uniques dans la table workshop
--        On prend les valeurs du premier statut rencontre pour chaque code (MIN(rowid))
INSERT INTO llx_workshop_operationorder_status (
    date_creation,
    entity,
    code,
    label,
    color,
    rang,
    status,
    planable,
    clean_event,
    display_on_planning,
    check_virtual_stock,
    or_pointable,
    save_date_cloture,
    require_planned_date,
    update_vehicule_info,
    require_conf,
    import_key
)
SELECT
    NOW(),
    0,
    old_s.code,
    old_s.label,
    IFNULL(NULLIF(old_s.color, ''), '#3c8dbc'),
    old_s.rang,
    CAST(IFNULL(NULLIF(old_s.status, ''), 1) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.planable, ''), 0) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.clean_event, ''), 0) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.display_on_planning, ''), 0) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.check_virtual_stock, ''), 0) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.or_pointable, ''), 0) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.save_date_cloture, ''), 0) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.require_planned_date, ''), 0) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.update_vehicule_info, ''), 0) AS UNSIGNED),
    CAST(IFNULL(NULLIF(old_s.require_conf, ''), 1) AS UNSIGNED),
    'mig-oo'
FROM llx_operationorder_status old_s
INNER JOIN (
    SELECT code, MIN(rowid) AS min_rowid
    FROM llx_operationorder_status
    GROUP BY code
) first_per_code ON old_s.rowid = first_per_code.min_rowid
WHERE old_s.code IS NOT NULL
  AND old_s.code != '';

-- 1.3 : Creation de la table de correspondance ancien status -> nouveau status
--        Chaque ancien rowid (toutes entites confondues) est mappe vers le nouveau rowid via le code
CREATE TABLE IF NOT EXISTS llx_migration_operationorder_status_map (
    old_status_id   integer NOT NULL,
    new_status_id   integer NOT NULL,
    code            varchar(128) NOT NULL,
    old_entity      integer NOT NULL DEFAULT 0,
    PRIMARY KEY (old_status_id)
) ENGINE=innodb;

-- 1.4 : Remplissage de la table de correspondance
INSERT INTO llx_migration_operationorder_status_map (old_status_id, new_status_id, code, old_entity)
SELECT
    old_s.rowid       AS old_status_id,
    new_s.rowid       AS new_status_id,
    old_s.code        AS code,
    old_s.entity      AS old_entity
FROM llx_operationorder_status old_s
INNER JOIN llx_workshop_operationorder_status new_s ON new_s.code = old_s.code
WHERE old_s.code IS NOT NULL
  AND old_s.code != '';

-- ============================================================================
-- ETAPE 2 : Migration des transitions de statuts (status_target)
-- ============================================================================
-- Strategie :
--   - Les anciennes transitions referencent les anciens rowid de statuts
--   - On remappe via llx_migration_operationorder_status_map
--   - Les transitions etaient dupliquees par entite, on dedoublonne avec DISTINCT
-- ============================================================================

-- 2.1 : Vidage de la table de destination avant insertion
TRUNCATE TABLE llx_workshop_operationorder_status_target;

-- 2.2 : Insertion des transitions avec remapping des rowid via la table de correspondance
INSERT INTO llx_workshop_operationorder_status_target (
    date_creation,
    fk_workshop_operationorderstatus,
    fk_workshop_operationorderstatus_target
)
SELECT DISTINCT
    NOW(),
    map_src.new_status_id,
    map_dst.new_status_id
FROM llx_operationorder_status_target old_t
INNER JOIN llx_migration_operationorder_status_map map_src
    ON map_src.old_status_id = old_t.fk_operationorderstatus
INNER JOIN llx_migration_operationorder_status_map map_dst
    ON map_dst.old_status_id = old_t.fk_operationorderstatus_target;

-- ============================================================================
-- ETAPE 3 : Migration des droits par groupe utilisateur (status_usergroup_rights)
-- ============================================================================
-- Strategie :
--   - Table geree par entite : on conserve l'entite d'origine
--   - L'ancienne table n'avait pas de champ entity, on le recupere depuis
--     la table de correspondance (old_entity du statut source)
--   - On remappe fk_operationorderstatus via la table de correspondance
-- ============================================================================

-- 3.1 : Vidage de la table de destination avant insertion
TRUNCATE TABLE llx_workshop_operationorder_status_usergroup_rights;

-- 3.2 : Insertion des droits avec remapping du statut et recuperation de l'entite
INSERT INTO llx_workshop_operationorder_status_usergroup_rights (
    date_creation,
    fk_workshop_operationorderstatus,
    code,
    fk_usergroup,
    entity
)
SELECT
    NOW(),
    map_s.new_status_id,
    old_r.code,
    old_r.fk_usergroup,
    map_s.old_entity
FROM llx_operationorder_status_usergroup_rights old_r
INNER JOIN llx_migration_operationorder_status_map map_s
    ON map_s.old_status_id = old_r.fk_operationorderstatus
INNER JOIN llx_usergroup ug
    ON ug.rowid = old_r.fk_usergroup;

-- ============================================================================
-- ETAPE 4 : Migration des conducteurs
-- ============================================================================
-- Strategie :
--   - Les conducteurs etaient des contacts (llx_socpeople) identifies par
--     un extrafield "driver" = 1 dans llx_socpeople_extrafields
--   - On les migre vers la table dediee llx_workshop_conducteur
--   - On force le rowid du conducteur = rowid du contact d'origine
--     => pas besoin de table de correspondance, le fk_conducteur des OR
--        pointera directement vers le bon enregistrement
-- ============================================================================

-- 4.1 : Vidage de la table de destination avant insertion
TRUNCATE TABLE llx_workshop_conducteur;

-- 4.2 : Insertion des conducteurs en conservant le rowid d'origine
INSERT INTO llx_workshop_conducteur (
    rowid,
    date_creation,
    nom,
    prenom,
    fk_soc,
    fk_user_creat
)
SELECT
    sp.rowid,
    NOW(),
    sp.lastname,
    IFNULL(sp.firstname, ''),
    sp.fk_soc,
    NULL
FROM llx_socpeople sp
INNER JOIN llx_socpeople_extrafields spx
    ON spx.fk_object = sp.rowid
WHERE spx.driver = 1;

-- ============================================================================
-- ETAPE 8 : Mise a jour de la main d'oeuvre (MO) sur les jobs
-- ============================================================================
-- Strategie :
--   - Pour chaque job migre, on cherche ses lignes enfants de MO dans l'ancien module
--   - Une ligne est de la MO si le produit a l'extrafield or_scan = 1
--   - Les lignes enfants sont liees par fk_parent_line = rowid du job (ancien det)
--   - qty_mo = somme des quantites MO
--   - total_ht_mo = somme des total_ht des lignes MO
--   - prix_mo = total_ht_mo / qty_mo
-- ============================================================================

-- 8.1 : Mise a jour des jobs avec les donnees MO
UPDATE llx_workshop_operationorder_jobs wj
INNER JOIN (
    SELECT
        od.fk_parent_line AS job_rowid,
        SUM(od.qty) AS sum_qty,
        SUM(od.total_ht) AS sum_total_ht
    FROM llx_operationorderdet od
    INNER JOIN llx_product p ON p.rowid = od.fk_product
    INNER JOIN llx_product_extrafields pex ON pex.fk_object = p.rowid AND pex.or_scan = 1
    WHERE od.fk_parent_line IS NOT NULL
      AND od.fk_parent_line > 0
    GROUP BY od.fk_parent_line
) mo ON mo.job_rowid = wj.rowid
SET
    wj.qty_mo = mo.sum_qty,
    wj.total_ht_mo = mo.sum_total_ht,
    wj.prix_mo = CASE
        WHEN mo.sum_qty > 0 THEN mo.sum_total_ht / mo.sum_qty
        ELSE 0
    END;

-- 8.2 : Mise a jour des jobs par defaut (etape 7.3) avec les lignes MO orphelines
--        Ce sont les lignes MO (or_scan=1) sans parent (fk_parent_line NULL/0)
--        On les rattache au job par defaut de leur OR via fk_operation_order
UPDATE llx_workshop_operationorder_jobs wj
INNER JOIN (
    SELECT
        od.fk_operation_order AS or_rowid,
        SUM(od.qty) AS sum_qty,
        SUM(od.total_ht) AS sum_total_ht
    FROM llx_operationorderdet od
    INNER JOIN llx_product p ON p.rowid = od.fk_product
    INNER JOIN llx_product_extrafields pex ON pex.fk_object = p.rowid AND pex.or_scan = 1
    WHERE (od.fk_parent_line IS NULL OR od.fk_parent_line = 0)
    GROUP BY od.fk_operation_order
) mo ON mo.or_rowid = wj.fk_operationorder
SET
    wj.qty_mo = mo.sum_qty,
    wj.total_ht_mo = mo.sum_total_ht,
    wj.prix_mo = CASE
        WHEN mo.sum_qty > 0 THEN mo.sum_total_ht / mo.sum_qty
        ELSE 0
    END
WHERE wj.import_key LIKE 'mig-or-%';

-- 8.4 : Mise a jour de ces jobs avec les lignes MO orphelines

-- 8.5 : Concatenation des descriptions de MO dans les jobs
--        Cas 1 : Jobs migres (etape 7.2) — lignes MO enfants via fk_parent_line
UPDATE llx_workshop_operationorder_jobs wj
INNER JOIN (
    SELECT
        od.fk_parent_line AS job_rowid,
        GROUP_CONCAT(od.description SEPARATOR '\n') AS concat_desc
    FROM llx_operationorderdet od
    INNER JOIN llx_product p ON p.rowid = od.fk_product
    INNER JOIN llx_product_extrafields pex ON pex.fk_object = p.rowid AND pex.or_scan = 1
    WHERE od.fk_parent_line IS NOT NULL
      AND od.fk_parent_line > 0
      AND od.description IS NOT NULL
      AND od.description != ''
    GROUP BY od.fk_parent_line
) mo ON mo.job_rowid = wj.rowid
SET wj.description = CONCAT_WS('\n', wj.description, mo.concat_desc);

-- 8.6 : Cas 2 : Jobs par defaut des OR sans job (etape 7.3) — lignes MO orphelines
UPDATE llx_workshop_operationorder_jobs wj
INNER JOIN (
    SELECT
        od.fk_operation_order AS or_rowid,
        GROUP_CONCAT(od.description SEPARATOR '\n') AS concat_desc
    FROM llx_operationorderdet od
    INNER JOIN llx_product p ON p.rowid = od.fk_product
    INNER JOIN llx_product_extrafields pex ON pex.fk_object = p.rowid AND pex.or_scan = 1
    WHERE (od.fk_parent_line IS NULL OR od.fk_parent_line = 0)
      AND od.description IS NOT NULL
      AND od.description != ''
    GROUP BY od.fk_operation_order
) mo ON mo.or_rowid = wj.fk_operationorder
